\newcounter{bookchap}
\newcommand{\bookchap}[2]{%
  \stepcounter{bookchap}%
  \clearpage
  \thispagestyle{opening}%
  \phantomsection
  \addcontentsline{toc}{chapter}{\protect\numberline{#1}{#2}}%
  \pdfbookmark[1]{#1. #2}{chap#1}%
  \markboth{Chapitre #1.\hspace{0.55em}#2}{Chapitre #1.\hspace{0.55em}#2}%
  \vspace*{7mm}%
  {\par\raggedright
    {\sffamily\fontsize{45}{45}\selectfont\color{palecyan}#1\par}
    \vspace{1.2mm}
    {\sffamily\fontsize{20.5}{25.5}\selectfont\bfseries\color{navy}#2\par}
    \vspace{4.4mm}
    \accenterule\par
  }
  \par\vspace{10.5mm}%
}

\newcounter{secbook}
\newcommand{\booksection}[2]{%
  \clearpage
  \thispagestyle{opening}%
  \phantomsection
  \stepcounter{secbook}%
  \ifx\relax#1\relax
    \pdfbookmark[1]{#2}{sec\the\value{secbook}}%
    \addcontentsline{toc}{chapter}{#2}%
    \markboth{#2}{#2}%
  \else
    \pdfbookmark[1]{#1 — #2}{sec\the\value{secbook}}%
    \addcontentsline{toc}{chapter}{#1 — #2}%
    \markboth{#1 — #2}{#1 — #2}%
  \fi
  \vspace*{7mm}%
  {\par\raggedright
    \ifx\relax#1\relax\else
      {\sffamily\footnotesize\bfseries\color{cyansea}\MakeUppercase{#1}\par}%
      \vspace{2.2mm}%
    \fi
    {\sffamily\fontsize{20.5}{25.5}\selectfont\bfseries\color{navy}#2\par}%
    \vspace{4.4mm}%
    \accenterule\par
  }
  \par\vspace{10.5mm}%
}

%  INTERCALAIRE DE PARTIE — pleine page navy, grille, constellation
% ============================================================
\newcommand{\partdivider}[2]{%
  \cleardoublepage
  \thispagestyle{emptynav}%
  \phantomsection
  \addcontentsline{toc}{part}{PARTIE #1 — #2}%
  \markboth{PARTIE #1 — #2}{PARTIE #1 — #2}%
  \begingroup
  \ifnum\pdfstrcmp{#1}{I}=0 \def\constx{24}\def\consty{120}\fi
  \ifnum\pdfstrcmp{#1}{II}=0 \def\constx{30}\def\consty{112}\fi
  \ifnum\pdfstrcmp{#1}{III}=0 \def\constx{22}\def\consty{120}\fi
  \ifnum\pdfstrcmp{#1}{IV}=0 \def\constx{32}\def\consty{118}\fi
  \ifnum\pdfstrcmp{#1}{V}=0 \def\constx{26}\def\consty{106}\fi
  \begin{tikzpicture}[remember picture, overlay]
    % fond
    \fill[navy] (current page.south west) rectangle (current page.north east);
    % grille
    \begin{scope}
      \clip (current page.south west) rectangle (current page.north east);
      \foreach \x in {0,14,...,148}
        {\draw[gridline] ($(current page.south west)+(\x mm,0)$) -- ($(current page.south west)+(\x mm,210mm)$);}
      \foreach \y in {0,14,...,210}
        {\draw[gridline] ($(current page.south west)+(0,\y mm)$) -- ($(current page.south west)+(148,\y mm)$);}
    \end{scope}
    % constellation (coordonnées absolues ancrées sur la page — sans scope shift)
    \coordinate (cA) at ($(current page.south west)+({\constx mm+0mm},{\consty mm+7mm})$);
    \coordinate (cB) at ($(current page.south west)+({\constx mm+12mm},{\consty mm+9mm})$);
    \coordinate (cC) at ($(current page.south west)+({\constx mm+24mm},{\consty mm+7mm})$);
    \coordinate (cD) at ($(current page.south west)+({\constx mm+33mm},{\consty mm+0mm})$);
    \coordinate (cE) at ($(current page.south west)+({\constx mm+26mm},{\consty mm-9mm})$);
    \coordinate (cF) at ($(current page.south west)+({\constx mm+12mm},{\consty mm-11mm})$);
    \coordinate (cG) at ($(current page.south west)+({\constx mm+1mm},{\consty mm-7mm})$);
    \coordinate (cH) at ($(current page.south west)+({\constx mm+41mm},{\consty mm+4mm})$);
    \coordinate (cI) at ($(current page.south west)+({\constx mm-8mm},{\consty mm+12mm})$);
    \draw[constline] (cA) -- (cB) -- (cC) -- (cD) -- (cE) -- (cF) -- (cG) -- (cA);
    \draw[constline] (cB) -- (cF);
    \draw[constline] (cC) -- (cF);
    \draw[constline] (cD) -- (cH);
    \draw[constline] (cA) -- (cI);
    \node[constnode] at (cA) {};  \node[constnode] at (cB) {};
    \node[constnode] at (cC) {};  \node[constnode] at (cD) {};
    \node[constnode] at (cE) {};  \node[constnode] at (cF) {};
    \node[constnode] at (cG) {};  \node[constnode] at (cH) {};
    \node[constnode] at (cI) {};
    \node[constnodeA] at (cF) {};
    \node[constnodeA] at (cC) {};
    % bloc titre
    \node[anchor=south west, text width=100mm] at ($(current page.south west)+(22mm,72mm)$) {%
      {\sffamily\footnotesize\bfseries\color{cyaneon}\MakeUppercase{Partie #1}\par}%
      \vspace{3.4mm}%
      {\sffamily\fontsize{23}{28.5}\selectfont\bfseries\color{white}#2\par}%
      \vspace{4.6mm}%
      {\color{cyaneon}\rule{17mm}{1.15pt}}%
    };
  \end{tikzpicture}%
  \endgroup
  \clearpage
}
